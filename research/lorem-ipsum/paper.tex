\section{Introduction}

Take a square. Cut it into nine equal squares and throw away the middle one. Do the same to each of the eight survivors, and keep going. \Cref{fig:carpet} shows the second step: a square lace with $64$ cells where $81$ could stand. Mathematicians call the limit of this game the Sierpi\'nski carpet; here, for one paper only, it is the \emph{lorem carpet}.

\section{Definitions}

\begin{definition}
\label{def:carpet}
For an integer $L \ge 0$ and $0 \le x, y < 3^L$, write $x_i, y_i \in \{0, 1, 2\}$ for the $i$-th digits of $x$ and $y$ in base $3$. The \emph{level-$L$ lorem carpet} is
\[
C_L = \bigl\{ (x, y) : \text{no position } i \text{ has } x_i = y_i = 1 \bigr\}.
\]
\end{definition}

\begin{remark}
$C_1$ is the $3 \times 3$ grid with its center removed, since the center is the one cell with $x_0 = y_0 = 1$. Ut enim ad minim veniam, quis nostrud exercitation ullamco.
\end{remark}

\section{Results}

\begin{theorem}
\label{thm:count}
For every $L \ge 0$, the level-$L$ lorem carpet holds exactly $8^L$ cells: $\lvert C_L \rvert = 8^L$.
\end{theorem}

\begin{fact}
\label{fact:machine}
For $0 \le L \le 6$, direct enumeration of every pair $(x, y)$ with $0 \le x, y < 3^L$ returns $\lvert C_L \rvert = 8^L$ exactly: $1$, $8$, $64$, $512$, $4096$, $32768$, $262144$. Checked by \texttt{scripts/verify.py}.
\end{fact}

\section{Proofs}
\label{sec:proofs}

\begin{proof}[Proof of \cref{thm:count}]
A cell of $C_L$ is exactly a choice, in each of the $L$ digit positions, of a pair $(x_i, y_i) \in \{0, 1, 2\}^2 \setminus \{(1, 1)\}$. The positions are independent and each offers $9 - 1 = 8$ options, so $\lvert C_L \rvert = 8^L$.

Alternatively, by self-similarity: the leading digits $(x_{L-1}, y_{L-1})$ place a cell in one of the $8$ surviving cells of $C_1$, and the remaining digits range over a full copy of $C_{L-1}$ inside it. Hence $\lvert C_L \rvert = 8 \cdot \lvert C_{L-1} \rvert$ for $L \ge 1$, with $\lvert C_0 \rvert = 1$. Two proofs, one count: the first explains the formula, the second explains the picture.
\end{proof}

\section{Reproducibility}

Two scripts, plain Python~3 with no dependencies. \texttt{scripts/verify.py} runs in a few seconds: it enumerates every cell of $C_L$ for $0 \le L \le 6$, a $729 \times 729$ grid at the top, and asserts that the count equals $8^L$, dying loudly on any mismatch. \texttt{scripts/figure.py} redraws \cref{fig:carpet} from the same rule. The sequence $8^n$ is OEIS \href{https://oeis.org/A001018}{A001018} \cite{oeis}.
